Workflow name: "CI/CD — Build and Deploy"

Triggers:
  push: branches: [main]
  pull_request: branches: [main]

Environment variables (top-level env block):
  REGISTRY: ghcr.io/tchspprtcv/lexcv
  IMAGE_TAG: ${{ github.sha }}

Job 1 — build-and-push:
  runs-on: ubuntu-latest
  permissions:
    contents: read
    packages: write

  Steps:
  1. actions/checkout@v4
  2. Set up QEMU — docker/setup-qemu-action@v3
  3. Set up Docker Buildx — docker/setup-buildx-action@v3
  4. Cache Maven ~/.m2/repository:
       uses: actions/cache@v4
       key: ${{ runner.os }}-maven-${{ hashFiles('backend/pom.xml') }}
       restore-keys: ${{ runner.os }}-maven-
       path: ~/.m2/repository
  5. Cache pnpm store:
       uses: actions/cache@v4
       id: pnpm-cache
       key: ${{ runner.os }}-pnpm-${{ hashFiles('web/pnpm-lock.yaml') }}
       restore-keys: ${{ runner.os }}-pnpm-
       path: ~/.local/share/pnpm/store
  6. Log in to GHCR:
       uses: docker/login-action@v3
       with: registry: ghcr.io, username: ${{ github.actor }}, password: ${{ secrets.GITHUB_TOKEN }}
  7. Build and push backend image:
       uses: docker/build-push-action@v6
       with:
         context: ./backend
         file: ./backend/Dockerfile
         platforms: linux/amd64
         push: ${{ github.ref == 'refs/heads/main' }}
         tags: |
           ${{ env.REGISTRY }}/backend:latest
           ${{ env.REGISTRY }}/backend:${{ env.IMAGE_TAG }}
         cache-from: type=gha,scope=backend
         cache-to: type=gha,mode=max,scope=backend
  8. Build and push frontend image:
       uses: docker/build-push-action@v6
       with:
         context: ./web
         file: ./web/Dockerfile
         platforms: linux/amd64
         push: ${{ github.ref == 'refs/heads/main' }}
         tags: |
           ${{ env.REGISTRY }}/frontend:latest
           ${{ env.REGISTRY }}/frontend:${{ env.IMAGE_TAG }}
         build-args: |
           NEXT_PUBLIC_API_BASE_PATH=/api/v1
           BACKEND_API_ORIGIN=http://backend:8080
         cache-from: type=gha,scope=frontend
         cache-to: type=gha,mode=max,scope=frontend

Job 2 — deploy:
  needs: [build-and-push]
  runs-on: ubuntu-latest
  if: github.ref == 'refs/heads/main'

  Steps:
  1. Deploy via SSH:
       uses: appleboy/ssh-action@v1.2.0
       with:
         host: ${{ secrets.VPS_HOST }}
         username: ${{ secrets.VPS_USER }}
         key: ${{ secrets.VPS_SSH_KEY }}
         port: ${{ secrets.VPS_PORT }}
         script: |
           cd /opt/lexcv
           echo ${{ secrets.GHCR_PAT }} | docker login ghcr.io -u tchspprtcv --password-stdin
           export REGISTRY=ghcr.io/tchspprtcv/lexcv
           export IMAGE_TAG=latest
           docker compose -f docker-compose.yml -f docker-compose.prod.yml pull
           docker compose -f docker-compose.yml -f docker-compose.prod.yml up -d --no-build
           docker image prune -f

Note: IMAGE_TAG=latest on VPS because docker-compose.prod.yml default tag is latest and the :latest tag is always updated by the push step.
  </action>

<threat_model>
## Trust Boundaries

| Boundary | Description |
|---|---|
| GitHub runner → GHCR | Runner pushes images using GITHUB_TOKEN (scoped to the workflow) |
| GitHub runner → VPS | SSH connection using VPS_SSH_KEY secret; commands run as VPS_USER |
| VPS → GHCR | docker login using GHCR_PAT secret to pull images |
| GHCR_PAT in SSH script | PAT echoed via pipe into docker login stdin — not visible in logs if not printed |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|---|---|---|---|---|
| T-39-01 | Tampering | docker images pushed to GHCR | mitigate | Images are built from pinned Dockerfiles in the repo; GITHUB_TOKEN write scope is limited to packages on this workflow only (`permissions: packages: write`) |
| T-39-02 | Information Disclosure | VPS_SSH_KEY exposed in logs | mitigate | Use `secrets.VPS_SSH_KEY` directly as `key:` input to appleboy/ssh-action — never echo or print the secret; appleboy masks it automatically |
| T-39-03 | Information Disclosure | GHCR_PAT echoed in SSH script | mitigate | Pipe PAT via `--password-stdin` (not `-p` flag) so it never appears in the process list; GitHub Actions masks all `secrets.*` references in log output |
| T-39-04 | Elevation of Privilege | Arbitrary code execution on VPS via tampered workflow | mitigate | Require branch protection on `main` (require PR + review before merge) so that only approved commits trigger deploy; workflow `if: github.ref == 'refs/heads/main'` restricts deploy to main only |
| T-39-05 | Denial of Service | Runaway docker images filling VPS disk | mitigate | `docker image prune -f` runs after each deploy to remove dangling images |
| T-39-SC | Tampering | npm/pip/cargo/action version pinning | mitigate | All GitHub Actions are pinned to explicit versions (actions/checkout@v4, docker/build-push-action@v6, appleboy/ssh-action@v1.2.0) — no floating `@latest` tags |
</threat_model>
